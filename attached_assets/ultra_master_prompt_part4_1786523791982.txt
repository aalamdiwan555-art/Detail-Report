```kotlin
package com.ultra.autodetector

import android.app.Application
import android.app.NotificationChannel
import android.app.NotificationManager
import android.content.Context
import android.os.Build
import com.google.firebase.FirebaseApp
import com.google.firebase.firestore.FirebaseFirestore
import com.google.firebase.firestore.firestoreSettings
import com.google.firebase.firestore.persistentCacheSettings
import timber.log.Timber

class UltraAutoDetectorApp : Application() {

    companion object {
        const val NOTIFICATION_CHANNEL_ID = "detection_service_channel"
        const val NOTIFICATION_CHANNEL_NAME = "Detection Service"
        const val NOTIFICATION_CHANNEL_DESC = "Shows when screen detection is active"
    }

    override fun onCreate() {
        super.onCreate()
        FirebaseApp.initializeApp(this)
        val db = FirebaseFirestore.getInstance()
        val settings = firestoreSettings {
            setLocalCacheSettings(persistentCacheSettings {})
        }
        db.firestoreSettings = settings
        if (BuildConfig.DEBUG) {
            Timber.plant(Timber.DebugTree())
        }
        createNotificationChannel()
    }

    private fun createNotificationChannel() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                NOTIFICATION_CHANNEL_ID,
                NOTIFICATION_CHANNEL_NAME,
                NotificationManager.IMPORTANCE_LOW
            ).apply {
                description = NOTIFICATION_CHANNEL_DESC
                setShowBadge(false)
                enableLights(false)
                enableVibration(false)
            }
            val notificationManager = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
            notificationManager.createNotificationChannel(channel)
        }
    }
}
```

### 7.3 Telegram Helper (util/TelegramHelper.kt)

```kotlin
package com.ultra.autodetector.util

import android.content.Context
import android.content.Intent
import android.net.Uri
import android.os.Build
import timber.log.Timber
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

object TelegramHelper {

    fun openTelegramWithRenewalMessage(
        context: Context,
        userEmail: String,
        userStatus: String,
        userUid: String,
        deviceInfo: String
    ): Boolean {
        val currentTime = SimpleDateFormat("yyyy-MM-dd HH:mm:ss", Locale.getDefault())
            .format(Date())

        val message = Constants.TELEGRAM_MESSAGE_TEMPLATE.format(
            userEmail, userStatus.uppercase(), userUid, deviceInfo, currentTime
        )

        return try {
            val telegramIntent = Intent(Intent.ACTION_VIEW).apply {
                data = Uri.parse("tg://resolve?domain=dminofclicker&text=${Uri.encode(message)}")
                setPackage("org.telegram.messenger")
                flags = Intent.FLAG_ACTIVITY_NEW_TASK
            }

            if (telegramIntent.resolveActivity(context.packageManager) != null) {
                context.startActivity(telegramIntent)
                Timber.d("Opened Telegram app with renewal message")
                true
            } else {
                telegramIntent.setPackage("org.thunderdog.challegram")
                if (telegramIntent.resolveActivity(context.packageManager) != null) {
                    context.startActivity(telegramIntent)
                    Timber.d("Opened Telegram X with renewal message")
                    true
                } else {
                    openTelegramWeb(context, message)
                }
            }
        } catch (e: Exception) {
            Timber.e(e, "Failed to open Telegram app")
            openTelegramWeb(context, message)
        }
    }

    private fun openTelegramWeb(context: Context, message: String): Boolean {
        return try {
            val webIntent = Intent(Intent.ACTION_VIEW).apply {
                data = Uri.parse("https://t.me/dminofclicker?start=${Uri.encode(message)}")
                flags = Intent.FLAG_ACTIVITY_NEW_TASK
            }
            context.startActivity(webIntent)
            Timber.d("Opened Telegram web with renewal message")
            true
        } catch (e: Exception) {
            Timber.e(e, "Failed to open Telegram web")
            false
        }
    }

    fun isTelegramInstalled(context: Context): Boolean {
        val packages = listOf(
            "org.telegram.messenger",
            "org.thunderdog.challegram",
            "org.telegram.messenger.web"
        )
        return packages.any { packageName ->
            try {
                context.packageManager.getPackageInfo(packageName, 0)
                true
            } catch (e: Exception) {
                false
            }
        }
    }

    fun getDeviceInfo(): String {
        return "${Build.MANUFACTURER} ${Build.MODEL}, Android ${Build.VERSION.RELEASE}"
    }
}
```

### 7.4 Humanization Engine (util/HumanizationEngine.kt)

```kotlin
package com.ultra.autodetector.util

import kotlinx.coroutines.delay
import kotlin.random.Random

object HumanizationEngine {

    private var lastClickTime = 0L

    fun getMicroDelay(): Long {
        return Random.nextLong(
            Constants.MICRO_DELAY_MIN_MS,
            Constants.MICRO_DELAY_MAX_MS + 1
        )
    }

    fun applyJitter(targetX: Float, targetY: Float): Pair<Float, Float> {
        val jitterX = Random.nextInt(
            -Constants.JITTER_RANGE_PX,
            Constants.JITTER_RANGE_PX + 1
        ).toFloat()
        val jitterY = Random.nextInt(
            -Constants.JITTER_RANGE_PX,
            Constants.JITTER_RANGE_PX + 1
        ).toFloat()
        return Pair(targetX + jitterX, targetY + jitterY)
    }

    fun isCooldownPassed(): Boolean {
        val currentTime = System.currentTimeMillis()
        val timeSinceLastClick = currentTime - lastClickTime
        return timeSinceLastClick >= Constants.COOLDOWN_INTERVAL_MS
    }

    fun recordClick() {
        lastClickTime = System.currentTimeMillis()
    }

    fun resetCooldown() {
        lastClickTime = 0L
    }

    suspend fun humanizeClick(targetX: Float, targetY: Float): Pair<Float, Float>? {
        if (!isCooldownPassed()) {
            return null
        }
        val delay = getMicroDelay()
        kotlinx.coroutines.delay(delay)
        val (jitteredX, jitteredY) = applyJitter(targetX, targetY)
        recordClick()
        return Pair(jitteredX, jitteredY)
    }
}
```

```kotlin
}

    Snackbar.make(this, message, duration).show()

fun View.showSnackbarWithAction(
    message: String,
    actionText: String,
    duration: Int = Snackbar.LENGTH_LONG,
    action: () -> Unit
) {
    Snackbar.make(this, message, duration)
        .setAction(actionText) { action() }
        .show()
}

fun <T> LiveData<T>.observeOnce(lifecycleOwner: LifecycleOwner, observer: (T) -> Unit) {
    observe(lifecycleOwner, object : Observer<T> {
        override fun onChanged(value: T) {
            removeObserver(this)
            observer(value)
        }
    })
}
```

```kotlin
package com.ultra.autodetector.data.local

import android.content.Context
import android.content.SharedPreferences
import androidx.security.crypto.EncryptedSharedPreferences
import androidx.security.crypto.MasterKey
import com.ultra.autodetector.util.Constants
import timber.log.Timber

class EncryptedPrefsManager private constructor(context: Context) {

    private val masterKey: MasterKey = MasterKey.Builder(context)
        .setKeyScheme(MasterKey.KeyScheme.AES256_GCM)
        .build()

    private val prefs: SharedPreferences = EncryptedSharedPreferences.create(
        context,
        Constants.PREFS_FILE_NAME,
        masterKey,
        EncryptedSharedPreferences.PrefKeyEncryptionScheme.AES256_SIV,
        EncryptedSharedPreferences.PrefValueEncryptionScheme.AES256_GCM
    )

    companion object {
        @Volatile
        private var instance: EncryptedPrefsManager? = null

        fun getInstance(context: Context): EncryptedPrefsManager {
            return instance ?: synchronized(this) {
                instance ?: EncryptedPrefsManager(context.applicationContext).also {
                    instance = it
                }
            }
        }
    }

    fun isFirstLaunch(): Boolean {
        return prefs.getBoolean(Constants.KEY_IS_FIRST_LAUNCH, true)
    }

    fun setFirstLaunchComplete() {
        prefs.edit().putBoolean(Constants.KEY_IS_FIRST_LAUNCH, false).apply()
    }

    fun isAccessibilityGranted(): Boolean {
        return prefs.getBoolean(Constants.KEY_ACCESSIBILITY_GRANTED, false)
    }

    fun setAccessibilityGranted(granted: Boolean) {
        prefs.edit().putBoolean(Constants.KEY_ACCESSIBILITY_GRANTED, granted).apply()
    }

    fun isOverlayGranted(): Boolean {
        return prefs.getBoolean(Constants.KEY_OVERLAY_GRANTED, false)
    }

    fun setOverlayGranted(granted: Boolean) {
        prefs.edit().putBoolean(Constants.KEY_OVERLAY_GRANTED, granted).apply()
    }

    fun isMediaProjectionGranted(): Boolean {
        return prefs.getBoolean(Constants.KEY_MEDIA_PROJECTION_GRANTED, false)
    }

    fun setMediaProjectionGranted(granted: Boolean) {
        prefs.edit().putBoolean(Constants.KEY_MEDIA_PROJECTION_GRANTED, granted).apply()
    }

    fun saveMediaProjectionData(resultCode: Int, data: android.content.Intent) {
        prefs.edit()
            .putInt(Constants.KEY_MEDIA_PROJECTION_RESULT, resultCode)
            .putString(Constants.KEY_MEDIA_PROJECTION_DATA, data.toUri(0))
            .apply()
    }

    fun getMediaProjectionResultCode(): Int {
        return prefs.getInt(Constants.KEY_MEDIA_PROJECTION_RESULT, -1)
    }

    fun getMediaProjectionData(): android.content.Intent? {
        val uriString = prefs.getString(Constants.KEY_MEDIA_PROJECTION_DATA, null) ?: return null
        return try {
            android.content.Intent.parseUri(uriString, 0)
        } catch (e: Exception) {
            Timber.e(e, "Failed to parse MediaProjection intent")
            null
        }
    }

    fun clearAll() {
        prefs.edit().clear().apply()
    }
}
```
