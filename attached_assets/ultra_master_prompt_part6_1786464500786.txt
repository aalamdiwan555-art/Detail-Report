### 10.1 OpenCV Manager (opencv/OpenCvManager.kt)

```kotlin
package com.ultra.autodetector.opencv

import android.graphics.Bitmap
import android.graphics.ImageFormat
import android.media.Image
import android.media.ImageReader
import org.opencv.android.Utils
import org.opencv.core.Core
import org.opencv.core.CvType
import org.opencv.core.Mat
import org.opencv.core.Point
import org.opencv.imgproc.Imgproc
import timber.log.Timber
import java.nio.ByteBuffer

/**
 * Central manager for all OpenCV operations.
 * Handles image conversion, template matching, and coordinate extraction.
 */
object OpenCvManager {

    /**
     * Converts Android Image (from ImageReader) to OpenCV Mat.
     * Supports RGBA_8888 format.
     */
    fun imageToMat(image: Image): Mat {
        val width = image.width
        val height = image.height
        val planes = image.planes
        val buffer = planes[0].buffer

        val pixelStride = planes[0].pixelStride
        val rowStride = planes[0].rowStride
        val rowPadding = rowStride - pixelStride * width

        // Create Mat from buffer
        val mat = Mat(height, width, CvType.CV_8UC4)
        val bytes = ByteArray(buffer.capacity())
        buffer.get(bytes)
        mat.put(0, 0, bytes)

        return mat
    }

    /**
     * Converts Bitmap to OpenCV Mat.
     */
    fun bitmapToMat(bitmap: Bitmap): Mat {
        val mat = Mat()
        Utils.bitmapToMat(bitmap, mat)
        return mat
    }

    /**
     * Converts OpenCV Mat to Bitmap.
     */
    fun matToBitmap(mat: Mat): Bitmap {
        val bitmap = Bitmap.createBitmap(mat.cols(), mat.rows(), Bitmap.Config.ARGB_8888)
        Utils.matToBitmap(mat, bitmap)
        return bitmap
    }

    /**
     * Creates ImageReader for screen capture with optimal settings.
     */
    fun createImageReader(width: Int, height: Int): ImageReader {
        return ImageReader.newInstance(
            width,
            height,
            ImageFormat.RGBA_8888,
            2  // Max images to prevent buffer overflow
        )
    }

    /**
     * Releases Mat resources to prevent memory leaks.
     */
    fun releaseMat(mat: Mat?) {
        mat?.release()
    }

    /**
     * Clones a Mat safely.
     */
    fun cloneMat(mat: Mat): Mat {
        return mat.clone()
    }
}
```

### 10.2 Screen Capture Manager (opencv/ScreenCaptureManager.kt)

```kotlin
package com.ultra.autodetector.opencv

import android.app.Activity
import android.content.Context
import android.content.Intent
import android.hardware.display.DisplayManager
import android.hardware.display.VirtualDisplay
import android.media.ImageReader
import android.media.projection.MediaProjection
import android.media.projection.MediaProjectionManager
import android.os.Handler
import android.os.Looper
import android.util.DisplayMetrics
import android.view.WindowManager
import org.opencv.core.Mat
import timber.log.Timber

/**
 * Manages screen capture using MediaProjection API.
 * Handles VirtualDisplay creation, ImageReader setup, and frame acquisition.
 */
class ScreenCaptureManager(private val context: Context) {

    private var mediaProjection: MediaProjection? = null
    private var virtualDisplay: VirtualDisplay? = null
    private var imageReader: ImageReader? = null
    private val handler = Handler(Looper.getMainLooper())

    private val windowManager = context.getSystemService(Context.WINDOW_SERVICE) as WindowManager
    private val displayMetrics = DisplayMetrics()

    val screenWidth: Int
        get() = displayMetrics.widthPixels

    val screenHeight: Int
        get() = displayMetrics.heightPixels

    val density: Int
        get() = displayMetrics.densityDpi

    init {
        windowManager.defaultDisplay.getMetrics(displayMetrics)
    }

    /**
     * Starts screen capture with MediaProjection.
     * Must be called after obtaining MediaProjection token from user.
     */
    fun startCapture(resultCode: Int, resultData: Intent, onFrameAvailable: (Mat) -> Unit) {
        val mediaProjectionManager = context.getSystemService(Context.MEDIA_PROJECTION_SERVICE) 
            as MediaProjectionManager

        mediaProjection = mediaProjectionManager.getMediaProjection(resultCode, resultData)

        imageReader = OpenCvManager.createImageReader(screenWidth, screenHeight)

        virtualDisplay = mediaProjection?.createVirtualDisplay(
            "ScreenCapture",
            screenWidth,
            screenHeight,
            density,
            DisplayManager.VIRTUAL_DISPLAY_FLAG_AUTO_MIRROR,
            imageReader?.surface,
            null,
            handler
        )

        // Set up frame listener
        imageReader?.setOnImageAvailableListener({ reader ->
            val image = reader.acquireLatestImage()
            image?.use { img ->
                try {
                    val mat = OpenCvManager.imageToMat(img)
                    onFrameAvailable(mat)
                    OpenCvManager.releaseMat(mat)
                } catch (e: Exception) {
                    Timber.e(e, "Error processing captured frame")
                }
            }
        }, handler)

        Timber.d("Screen capture started: ${screenWidth}x$screenHeight")
    }

    /**
     * Stops screen capture and releases all resources.
     */
    fun stopCapture() {
        virtualDisplay?.release()
        virtualDisplay = null

        imageReader?.setOnImageAvailableListener(null, null)
        imageReader?.close()
        imageReader = null

        mediaProjection?.stop()
        mediaProjection = null

        Timber.d("Screen capture stopped")
    }

    /**
     * Checks if screen capture is currently active.
     */
    fun isCapturing(): Boolean {
        return mediaProjection != null && virtualDisplay != null
    }

    /**
     * Gets the latest captured frame as Mat.
     * Returns null if no frame is available.
     */
    fun getLatestFrame(): Mat? {
        val image = imageReader?.acquireLatestImage() ?: return null
        return try {
            val mat = OpenCvManager.imageToMat(image)
            image.close()
            mat
        } catch (e: Exception) {
            image.close()
            Timber.e(e, "Error getting latest frame")
            null
        }
    }
}
```

### 10.3 Template Matcher (opencv/TemplateMatcher.kt)

```kotlin
package com.ultra.autodetector.opencv

import com.ultra.autodetector.util.Constants
import org.opencv.core.Core
import org.opencv.core.CvType
import org.opencv.core.Mat
import org.opencv.core.Point
import org.opencv.imgproc.Imgproc
import timber.log.Timber
import kotlin.math.max
import kotlin.math.min

/**
 * Performs OpenCV template matching on screen captures.
 * Uses normalized cross-correlation coefficient matching.
 */
class TemplateMatcher {

    data class MatchResult(
        val found: Boolean,
        val confidence: Double,
        val centerX: Float,
        val centerY: Float,
        val topLeft: Point,
        val bottomRight: Point,
        val templateWidth: Int,
        val templateHeight: Int
    )

    /**
     * Matches a single template against the screen Mat.
     * 
     * @param screenMat The captured screen as OpenCV Mat (RGBA)
     * @param templateMat The template image as OpenCV Mat
     * @param threshold Minimum confidence threshold (default: 0.85)
     * @return MatchResult containing match details
     */
    fun matchTemplate(
        screenMat: Mat,
        templateMat: Mat,
        threshold: Double = Constants.CONFIDENCE_THRESHOLD
    ): MatchResult {

        // Validate dimensions
        if (templateMat.cols() > screenMat.cols() || templateMat.rows() > screenMat.rows()) {
            Timber.w("Template larger than screen, skipping match")
            return MatchResult(false, 0.0, 0f, 0f, Point(0.0, 0.0), Point(0.0, 0.0), 0, 0)
        }

        // Create result matrix
        val resultCols = screenMat.cols() - templateMat.cols() + 1
        val resultRows = screenMat.rows() - templateMat.rows() + 1
        val resultMat = Mat(resultRows, resultCols, CvType.CV_32FC1)

        try {
            // Perform template matching using normalized cross-correlation
            Imgproc.matchTemplate(
                screenMat,
                templateMat,
                resultMat,
                Imgproc.TM_CCOEFF_NORMED
            )

            // Find maximum value location
            val minMaxLoc = Core.minMaxLoc(resultMat)
            val maxVal = minMaxLoc.maxVal
            val maxLoc = minMaxLoc.maxLoc

            // Check if match exceeds threshold
            val found = maxVal >= threshold

            if (found) {
                // Calculate center coordinates
                val centerX = (maxLoc.x + templateMat.cols() / 2.0).toFloat()
                val centerY = (maxLoc.y + templateMat.rows() / 2.0).toFloat()

                val bottomRight = Point(
                    maxLoc.x + templateMat.cols(),
                    maxLoc.y + templateMat.rows()
                )

                Timber.d("Match found! Confidence: %.3f at (%.1f, %.1f)", 
                    maxVal, centerX, centerY)

                return MatchResult(
                    found = true,
                    confidence = maxVal,
                    centerX = centerX,
                    centerY = centerY,
                    topLeft = maxLoc,
                    bottomRight = bottomRight,
                    templateWidth = templateMat.cols(),
                    templateHeight = templateMat.rows()
                )
            }

            return MatchResult(
                found = false,
                confidence = maxVal,
                centerX = 0f,
                centerY = 0f,
                topLeft = Point(0.0, 0.0),
                bottomRight = Point(0.0, 0.0),
                templateWidth = templateMat.cols(),
                templateHeight = templateMat.rows()
            )

        } finally {
            // Always release result matrix
            OpenCvManager.releaseMat(resultMat)
        }
    }

    /**
     * Matches multiple templates against the screen and returns all results.
     * 
     * @param screenMat The captured screen
     * @param templates List of template Mats
     * @param threshold Minimum confidence threshold
     * @return List of MatchResult for all templates
     */
    fun matchMultipleTemplates(
        screenMat: Mat,
        templates: List<Mat>,
        threshold: Double = Constants.CONFIDENCE_THRESHOLD
    ): List<MatchResult> {
        return templates.mapIndexed { index, template ->
            val result = matchTemplate(screenMat, template, threshold)
            if (result.found) {
                Timber.d("Template $index matched with confidence %.3f", result.confidence)
            }
            result
        }
    }

    /**
     * Finds the best match among multiple templates.
     * 
     * @param screenMat The captured screen
     * @param templates List of template Mats
     * @param threshold Minimum confidence threshold
     * @return Best MatchResult, or null if no match exceeds threshold
     */
    fun findBestMatch(
        screenMat: Mat,
        templates: List<Mat>,
        threshold: Double = Constants.CONFIDENCE_THRESHOLD
    ): MatchResult? {
        val results = matchMultipleTemplates(screenMat, templates, threshold)
        return results.filter { it.found }.maxByOrNull { it.confidence }
    }

    /**
     * Preprocesses template image for optimal matching.
     * Converts to RGBA if needed and normalizes.
     */
    fun preprocessTemplate(templateMat: Mat): Mat {
        val processed = Mat()

        // Ensure 4-channel (RGBA)
        if (templateMat.channels() == 3) {
            Imgproc.cvtColor(templateMat, processed, Imgproc.COLOR_RGB2RGBA)
        } else if (templateMat.channels() == 1) {
            Imgproc.cvtColor(templateMat, processed, Imgproc.COLOR_GRAY2RGBA)
        } else {
            templateMat.copyTo(processed)
        }

        return processed
    }

    /**
     * Loads template from file path and preprocesses it.
     */
    fun loadTemplateFromFile(path: String): Mat? {
        return try {
            val mat = org.opencv.imgcodecs.Imgcodecs.imread(path, org.opencv.imgcodecs.Imgcodecs.IMREAD_UNCHANGED)
            if (mat.empty()) {
                Timber.w("Failed to load template from: $path")
                null
            } else {
                preprocessTemplate(mat)
            }
        } catch (e: Exception) {
            Timber.e(e, "Error loading template: $path")
            null
        }
    }
}
```
