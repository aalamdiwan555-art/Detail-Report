## 📋 EXECUTIVE SUMMARY

Build a **production-ready Android application** in **Kotlin** that performs real-time screen detection using **OpenCV template matching**, **MediaProjection API** for screen capture, **AccessibilityService** for humanized auto-clicking, and **Firebase** (Auth, Firestore, Cloud Storage) for user management, license control, and template distribution. Includes a **Telegram integration** for account renewal requests with pre-filled admin contact and user data.

**Target API Level:** 33+ (Android 13+)  
**Minimum API Level:** 26 (Android 8.0)  
**Language:** Kotlin  
**Architecture:** MVVM + Service-Oriented

service cloud.firestore {

    // USER ACCOUNTS COLLECTION

    match /users/{userId} {
      // READ: User can read own document OR admin can read all
      allow read: if request.auth != null 
        && (request.auth.uid == userId 
            || request.auth.token.email == 'diwanatik84@gmail.com');

      // CREATE: Any authenticated user can create their own document
      allow create: if request.auth != null 
        && request.auth.uid == userId
        && request.resource.data.role == 'user'
        && request.resource.data.status == 'pending'
        && request.resource.data.expirationTimestamp == null;

      // WRITE: ONLY admin can update any user document
      allow write: if request.auth != null 
        && request.auth.token.email == 'diwanatik84@gmail.com';
    }

    // ============================================================

      // READ: Any authenticated user can read templates

      // WRITE: ONLY admin can create/update/delete templates

### 2.1 User Data Model (data/model/User.kt)

```kotlin
package com.ultra.autodetector.data.model

import com.google.firebase.Timestamp
import com.google.firebase.firestore.PropertyName

/**
 * Represents a user account in the Firestore database.
 * 
 * @property uid Firebase Authentication UID
 * @property email User's registered email address
 * @property role User role: "user" or "admin"
 * @property status Account status: "pending", "approved", "rejected", "expired"
 * @property expirationTimestamp License expiration time in milliseconds (null = never approved)
 * @property createdAt Account creation timestamp (server-side)
 * @property lastLoginAt Last successful login timestamp
 * @property deviceInfo Device model and Android version for tracking
 */
data class User(
    @PropertyName("uid") val uid: String = "",
    @PropertyName("email") val email: String = "",
    @PropertyName("role") val role: String = "user",
    @PropertyName("status") val status: String = "pending",
    @PropertyName("expirationTimestamp") val expirationTimestamp: Long? = null,
    @PropertyName("createdAt") val createdAt: Timestamp? = null,
    @PropertyName("lastLoginAt") val lastLoginAt: Timestamp? = null,
    @PropertyName("deviceInfo") val deviceInfo: String = ""
) {
    /**
     * Checks if the user's license is currently active.
     * Returns true if status is "approved" AND expiration is in the future.
     */
    fun isLicenseActive(): Boolean {
        if (status != "approved") return false
        val exp = expirationTimestamp ?: return false
        return System.currentTimeMillis() < exp
    }

    /**
     * Calculates remaining time in human-readable format.
     */
    fun getRemainingTimeText(): String {
        if (status == "lifetime" || expirationTimestamp == 4102444800000L) 
            return "Lifetime"
        val exp = expirationTimestamp ?: return "Not Approved"
        val diff = exp - System.currentTimeMillis()
        if (diff <= 0) return "Expired"
        val days = diff / (1000 * 60 * 60 * 24)
        val hours = (diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60)
        return "${days}d ${hours}h remaining"
    }
}
```

package com.ultra.autodetector.data.model

import com.google.firebase.Timestamp
import com.google.firebase.firestore.PropertyName

/**
 * Represents a detection template stored in Firestore and Firebase Storage.
 * 
 * @property templateId Unique template identifier (Firestore document ID)
 * @property name Human-readable template name
 * @property description Optional description of what this template detects
 * @property storagePath Firebase Storage path: "templates/{templateId}.png"
 * @property downloadUrl Public download URL for the template image
 * @property confidenceThreshold Override confidence threshold (default: 0.85)
 * @property isActive Whether this template is currently active for detection
 * @property createdAt Template creation timestamp
 * @property createdBy Admin UID who created this template
 */
data class Template(
    @PropertyName("templateId") val templateId: String = "",
    @PropertyName("name") val name: String = "",
    @PropertyName("description") val description: String = "",
    @PropertyName("storagePath") val storagePath: String = "",
    @PropertyName("downloadUrl") val downloadUrl: String = "",
    @PropertyName("confidenceThreshold") val confidenceThreshold: Double = 0.85,
    @PropertyName("isActive") val isActive: Boolean = true,
    @PropertyName("createdAt") val createdAt: Timestamp? = null,
    @PropertyName("createdBy") val createdBy: String = ""
)
